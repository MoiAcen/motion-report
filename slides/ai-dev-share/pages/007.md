<div class="cap">費用 = token 數 × 輸入單價。單價來源：claude.com/pricing（Haiku 5.5 取 100K tokens 以下）。effort 對 token 數的倍率為示意，非實際用量。</div>

<script>
(function () {
  var models = [
    { name: 'Haiku 5.5', price: 0.10 },
    { name: 'Sonnet 5.5', price: 2 },
    { name: 'Opus 5.5', price: 4 }
  ];
  // effort 倍率為示意值，用來說明 effort 會增加 token 數
  var efforts = [
    { name: 'low', mult: 1 },
    { name: 'medium', mult: 2 },
    { name: 'high', mult: 3 },
    { name: 'max', mult: 4 }
  ];
  var priceArr = models.map(function (m) { return m.price; });
  var multArr = efforts.map(function (e) { return e.mult; });
  var maxCost = priceArr[priceArr.length - 1] * multArr[multArr.length - 1];
  var mRange = document.getElementById('demo-model');
  if (!mRange) return;
  var eRange = document.getElementById('demo-effort');
  var mName = document.getElementById('demo-model-name');
  var eName = document.getElementById('demo-effort-name');
  var cost = document.getElementById('demo-cost');

  var mCur = 2, eCur = 2;   // 目前連續位置（檔位之間可以是小數）
  var mFrom = 2, eFrom = 2, mTo = 2, eTo = 2;
  var t0 = 0, DUR = 900;    // 每次漸進移動約 0.9 秒

  // 在連續位置 x 上取內插值，讓費用 bar 跟著滑桿平滑變化
  function lerpAt(arr, x) {
    var i = Math.min(Math.floor(x), arr.length - 1);
    var j = Math.min(i + 1, arr.length - 1);
    return arr[i] + (arr[j] - arr[i]) * (x - i);
  }
  function ease(t) { return t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2; }

  function render() {
    mRange.value = mCur;
    eRange.value = eCur;
    mName.textContent = models[Math.round(mCur)].name;
    eName.textContent = efforts[Math.round(eCur)].name;
    var c = lerpAt(priceArr, mCur) * lerpAt(multArr, eCur);
    cost.style.width = (c / maxCost * 100).toFixed(2) + '%';
  }

  function frame(now) {
    var t = Math.min(1, (now - t0) / DUR);
    var k = ease(t);
    mCur = mFrom + (mTo - mFrom) * k;
    eCur = eFrom + (eTo - eFrom) * k;
    render();
    if (t < 1) requestAnimationFrame(frame);
  }

  function randomize() {
    mFrom = mCur;
    eFrom = eCur;
    mTo = Math.floor(Math.random() * models.length);
    eTo = Math.floor(Math.random() * efforts.length);
    t0 = performance.now();
    requestAnimationFrame(frame);
  }

  render();
  randomize();
  setInterval(randomize, 1400);
})();
</script>

## 動效

- 眉標：fade-up，y 為 12px
- 標題：wipe，時長 mid
- 模型與 effort 滑桿：每 1.4 秒隨機選一個新檔位，滑桿與文字標籤以約 0.9 秒漸進移動（由腳本控制，不可拖曳）
- 費用 bar：寬度隨模型單價與 effort 倍率同步漸進變化
- mark：共用元素，與第 6 頁標題中「Token」字樣的 mark 相同 id，翻頁時從第 6 頁飛到「token 費用」標籤後面，成為底框；標籤文字為白色
